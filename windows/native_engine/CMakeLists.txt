cmake_minimum_required(VERSION 3.21)
project(TelexMiniNativeCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8 /W4 /EHsc)
endif()
# Shared engine works in both the standalone Windows workspace and the repo.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../native_core/core.cpp")
  set(SHARED_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../native_core")
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../native_core/core.cpp")
  set(SHARED_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../native_core")
else()
  message(FATAL_ERROR "Shared native_core is missing")
endif()
add_library(telexmini_native_core STATIC "${SHARED_CORE}/core.cpp")
target_include_directories(telexmini_native_core PUBLIC "${SHARED_CORE}")
target_compile_definitions(telexmini_native_core PUBLIC UNICODE _UNICODE NOMINMAX)
add_executable(telexmini_core_tests core_test.cpp)
target_link_libraries(telexmini_core_tests PRIVATE telexmini_native_core)
target_compile_definitions(telexmini_core_tests PRIVATE UNICODE _UNICODE NOMINMAX)
add_library(TelexMiniEngine SHARED core_bridge.cpp)
target_link_libraries(TelexMiniEngine PRIVATE telexmini_native_core)
target_compile_definitions(TelexMiniEngine PRIVATE UNICODE _UNICODE NOMINMAX)
